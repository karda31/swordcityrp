create table if not exists public.entries ( id bigint generated always as identity primary key, name text not null check (char_length(name) between 1 and 50), description text not null default '' check (char_length(description) <= 300), created_at timestamptz not null default now() ); alter table public.entries enable row level security; create policy "Public can view entries" on public.entries for select to anon, authenticated using (true); create policy "Public can add entries" on public.entries for insert to anon, authenticated with check ( char_length(name) between 1 and 50 and char_length(description) <= 300 ); alter publication supabase_realtime add table public.entries;
